### 必须验证的链路

1. **基础连接**
   - TCP 连接建立
   - `initialize` 握手
   - `tools/list` 获取工具列表
   - `resources/list` 获取资源列表

2. **MCP 操作**

3. **状态联动**（如果有测试数据）
   - 通过 MCP 安装应用 → GUI 列表刷新
   - 通过 MCP 卸载应用 → GUI 列表更新

4. **多客户端**
   - 同时连接两个客户端
   - 各自独立发送请求
   - 一个断开不影响另一个

5. **错误处理**
   - 发送格式错误的 JSON → 返回 parse error
   - 调用不存在的 method → 返回 method not found
   - 参数错误 → 返回 invalid params

### Step 1：编写 Python 冒烟测试脚本

**新建文件**：`Scripts/mcp_tcp_smoke.py`

```python
#!/usr/bin/env python3
"""TCP MCP Server 端到端冒烟测试"""

import socket
import json
import sys
import time

HOST = "127.0.0.1"
PORT = 19820

def send_request(sock, method, params=None, req_id=None):
    """发送 JSON-RPC 请求并等待响应"""
    if req_id is None:
        req_id = int(time.time() * 1000)
    
    request = {
        "jsonrpc": "2.0",
        "id": req_id,
        "method": method,
    }
    if params:
        request["params"] = params
    
    line = json.dumps(request) + "\n"
    sock.sendall(line.encode("utf-8"))
    
    # 读取响应（一行）
    response_data = b""
    while b"\n" not in response_data:
        chunk = sock.recv(4096)
        if not chunk:
            raise ConnectionError("Connection closed")
        response_data += chunk
    
    response_line = response_data.split(b"\n")[0]
    return json.loads(response_line)

def test_initialize(sock):
    """测试 MCP 初始化握手"""
    print("  Testing initialize...", end=" ")
    response = send_request(sock, "initialize", {
        "protocolVersion": "2024-11-05",
        "capabilities": {},
        "clientInfo": {"name": "smoke-test", "version": "1.0"}
    }, req_id=1)
    
    assert "result" in response, f"Expected result, got: {response}"
    assert "serverInfo" in response["result"], "Missing serverInfo"
    print("✅")
    return response

def test_tools_list(sock):
    """测试获取工具列表"""
    print("  Testing tools/list...", end=" ")
    response = send_request(sock, "tools/list", {}, req_id=2)
    
    assert "result" in response, f"Expected result, got: {response}"
    tools = response["result"].get("tools", [])
    assert len(tools) > 0, "Expected at least one tool"
    print(f"✅ ({len(tools)} tools)")
    return tools

def test_resources_list(sock):
    """测试获取资源列表"""
    print("  Testing resources/list...", end=" ")
    response = send_request(sock, "resources/list", {}, req_id=3)
    
    assert "result" in response, f"Expected result, got: {response}"
    print("✅")

def test_list_apps(sock):
    """测试列出应用"""
    print("  Testing tools/call list_apps...", end=" ")
    response = send_request(sock, "tools/call", {
        "name": "list_apps",
        "arguments": {}
    }, req_id=4)
    
    assert "result" in response, f"Expected result, got: {response}"
    print("✅")

def test_invalid_method(sock):
    """测试调用不存在的方法"""
    print("  Testing invalid method...", end=" ")
    response = send_request(sock, "nonexistent/method", {}, req_id=5)
    
    assert "error" in response, f"Expected error, got: {response}"
    print("✅")

def main():
    print(f"\n=== MCP TCP Smoke Test ===")
    print(f"Connecting to {HOST}:{PORT}...")
    
    try:
        sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        sock.settimeout(5.0)
        sock.connect((HOST, PORT))
        print("Connected!\n")
    except ConnectionRefusedError:
        print(f"❌ Connection refused. Is PlayCover.app running?")
        sys.exit(1)
    
    try:
        test_initialize(sock)
        test_tools_list(sock)
        test_resources_list(sock)
        test_list_apps(sock)
        test_invalid_method(sock)
        
        print(f"\n=== All tests passed! ===\n")
    except Exception as e:
        print(f"\n❌ Test failed: {e}")
        sys.exit(1)
    finally:
        sock.close()

if __name__ == "__main__":
    main()
```

```python
def test_multiple_clients():
    """测试多客户端并发"""
    print("  Testing multiple clients...", end=" ")
    
    sock1 = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    sock2 = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    sock1.settimeout(5.0)
    sock2.settimeout(5.0)
    
    sock1.connect((HOST, PORT))
    sock2.connect((HOST, PORT))
    
    # 两个客户端各自初始化
    r1 = send_request(sock1, "initialize", {...}, req_id=100)
    r2 = send_request(sock2, "initialize", {...}, req_id=200)
    
    assert "result" in r1 and "result" in r2
    
    # 断开一个，另一个仍然工作
    sock1.close()
    r3 = send_request(sock2, "tools/list", {}, req_id=201)
    assert "result" in r3
    
    sock2.close()
    print("✅")
```
